3. Memory Architecture: Union-Style (NaN-Boxing) & Stack-Style Execution

Developers frequently ask whether AAYU uses Union-Style or Stack-Style memory management. The engineering answer is: AAYU uses an optimized Hybrid Architecture combining Tagged Union (NaN-Boxing) for value representation with high-speed Call-Stack Frames for execution context.

Concept Samjhein (Hindi/Hinglish):

2. Execution Level (Stack Style): Har function call aur local variable stack frame par push hota hai aur function end hote hi O(1) speed me instant pop ho jata hai. Isliye local variables me garbage collector ki zaroorat nahi padti!

19.1 Linear Search (O(N) Time, O(1) Space)

action linear_search(arr, target)
    let i = 0
    let n = len(arr)
    while i < n
        if arr[i] == target
            return i # Found at index i
        end
        i = i + 1
    end
    return -1 # Not found
end

let numbers = [4, 8, 15, 16, 23, 42]

19.2 Binary Search (O(log N) Time, O(1) Space)

action binary_search(arr, target)
    let low = 0
    let high = len(arr) - 1
    while low <= high
        let mid = (low + high) / 2
        if arr[mid] == target
            return mid
        elif arr[mid] < target
            low = mid + 1
        else
            high = mid - 1
        end
    end
    return -1
end

let sorted_arr = [10, 25, 30, 45, 50, 60]
print("Binary search for 45:", binary_search(sorted_arr, 45)) # Output: 3

19.3 Bubble Sort (O(N2) Time, O(1) Space)

action bubble_sort(arr)
    let n = len(arr)
    let i = 0
    while i < n
        let j = 0
        while j < n - i - 1
            if arr[j] > arr[j + 1]
                let temp = arr[j]
                arr = set_at(arr, j, arr[j + 1])
                arr = set_at(arr, j + 1, temp)
            end
            j = j + 1
        end
        i = i + 1
    end
    return arr
end

let unsorted = [64, 34, 25, 12, 22, 11, 90]
let sorted = bubble_sort(unsorted)
print("Sorted Array:", sorted) # [11, 12, 22, 25, 34, 64, 90]

19.4 Stack Implementation (LIFO - Last In First Out)

action stack_push(s, val)

action stack_peek(s)

    return res

action max_subarray_sum_k(arr, k)
    let n = len(arr)
    require(k <= n, "Window size k cannot exceed array length")

        end
        j = j + 1
    end
